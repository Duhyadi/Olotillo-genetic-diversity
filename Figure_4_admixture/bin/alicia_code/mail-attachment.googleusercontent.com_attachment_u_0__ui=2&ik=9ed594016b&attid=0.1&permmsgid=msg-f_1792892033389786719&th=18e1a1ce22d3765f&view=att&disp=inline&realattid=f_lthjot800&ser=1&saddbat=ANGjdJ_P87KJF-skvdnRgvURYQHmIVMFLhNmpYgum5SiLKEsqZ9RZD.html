# generate same colors than the ones used for admixture plot
# I got this from the function that generates colors in pophelper when k>12 https://github.com/royfrancis/pophelper/blob/master/R/pophelper.R
piecolors<-gplots::rich.colors(13)

## plot background distribution
p.parvi<- ggplot() +
          geom_raster(data = raster_df , aes(x = x, y = y, fill =   factor(Zea_mays_parviglumis))) +
          scale_fill_manual(values = c("grey80", "grey10")) + 
          labs(x="", y= "")  +

### Add pies
         new_scale("fill") + # this is needed because raster and pie have diff fills
         geom_scatterpie(data=Qval,
                           aes(x=LONGITUDE, 
                               y=LATITUDE,
                               group=POB),
                               color=NA,
                               cols=paste0("K", 1:13)) +
 scale_fill_manual(values= piecolors,
                    name="Genetic cluster") + 
  theme_bw() + theme(legend.position="none") +
  theme(text = element_text(size = 20))  
p.parvi

# save file
ggsave("../data/genetic/admixture_outs/plot_map_parvi.png",
       plot=p.parvi, dpi=300,
       width=15, height = 11, units="cm")
